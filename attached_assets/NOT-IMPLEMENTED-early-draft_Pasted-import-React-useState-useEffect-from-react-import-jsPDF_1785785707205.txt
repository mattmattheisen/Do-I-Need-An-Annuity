import React, { useState, useEffect } from 'react';
import jsPDF from 'jspdf';

const BulwarkAnnuitySuitability = () => {
  const [step, setStep] = useState(1);
  const [formData, setFormData] = useState({
    age: '',
    employmentIncome: 0,
    selfEmploymentIncome: 0,
    socialSecurityAnnual: 0,
    pensionAnnual: 0,
    otherGuaranteedIncome: 0,
    investableAssets: 0,
    annualSpending: '',
    lifeExpectancy: '',
    heirsImportance: 'moderate',
    healthcareConcern: 'no',
    marketDownturnResponse: '',
  });

  const [results, setResults] = useState(null);
  const [mortTableToggle, setMortTableToggle] = useState('iam2012');

  useEffect(() => {
    window.scrollTo(0, 0);
  }, [step]);

  const handleChange = (e) => {
    const { name, value, type } = e.target;
    setFormData(prev => ({
      ...prev,
      [name]: type === 'number' ? parseFloat(value) || 0 : value
    }));
  };

  const getPayoutRate = (age) => {
    const rates = {
      50: 0.045, 52: 0.058, 55: 0.062, 60: 0.070, 65: 0.077,
      70: 0.092, 75: 0.110, 80: 0.135, 85: 0.175
    };
    if (rates[age]) return rates[age];
    const ages = Object.keys(rates).map(Number);
    const lower = ages.filter(a => a <= age).pop();
    const upper = ages.filter(a => a > age)[0];
    if (!upper) return rates[lower];
    const ratio = (age - lower) / (upper - lower);
    return rates[lower] + ratio * (rates[upper] - rates[lower]);
  };

  const calculateScore = () => {
    const age = parseInt(formData.age);
    const totalGuaranteedIncome = formData.socialSecurityAnnual + 
                                  formData.pensionAnnual + 
                                  formData.otherGuaranteedIncome;
    const annualSpending = parseFloat(formData.annualSpending);
    const lifeExpectancy = parseInt(formData.lifeExpectancy);

    // LONGEVITY COMPONENT (0-25)
    const planningHorizon = lifeExpectancy - age;
    let longevityScore = 0;
    if (planningHorizon < 10) longevityScore = 5;
    else if (planningHorizon < 15) longevityScore = 10;
    else if (planningHorizon < 20) longevityScore = 18;
    else longevityScore = 25;

    // INCOME GAP COMPONENT (0-25)
    const incomeGap = Math.max(0, annualSpending - totalGuaranteedIncome);
    const gapPercentage = incomeGap / annualSpending;
    let incomeGapScore = 0;
    if (gapPercentage === 0) incomeGapScore = 0;
    else if (gapPercentage < 0.15) incomeGapScore = 8;
    else if (gapPercentage < 0.30) incomeGapScore = 15;
    else if (gapPercentage < 0.50) incomeGapScore = 22;
    else incomeGapScore = 25;

    // FLEXIBILITY COMPONENT (0-25)
    let flexibilityScore = 25;
    if (formData.heirsImportance === 'high') flexibilityScore -= 10;
    if (formData.healthcareConcern === 'yes') flexibilityScore -= 10;
    flexibilityScore = Math.max(0, flexibilityScore);

    // BEHAVIORAL FIT COMPONENT (0-25)
    let behavioralScore = 0;
    if (formData.marketDownturnResponse === 'hold') behavioralScore = 13;
    else if (formData.marketDownturnResponse === 'reduce') behavioralScore = 10;
    else if (formData.marketDownturnResponse === 'increaseHoldings') behavioralScore = 5;
    else behavioralScore = 0;

    const totalScore = longevityScore + incomeGapScore + flexibilityScore + behavioralScore;

    // RECOMMENDATION CAP
    const investableAssets = parseFloat(formData.investableAssets) || 0;
    const maxAnnuityAmount = investableAssets * 0.50;
    const payoutRate = getPayoutRate(age);
    const maxRecommendedByIncome = incomeGap / payoutRate;
    const recommendedAmount = Math.min(maxRecommendedByIncome, maxAnnuityAmount);
    const recommendedPercentage = investableAssets > 0 
      ? (recommendedAmount / investableAssets) * 100 
      : 0;

    const estimatedAnnualIncome = recommendedAmount * payoutRate;

    return {
      totalScore: Math.round(totalScore),
      longevityScore,
      incomeGapScore,
      flexibilityScore,
      behavioralScore,
      incomeGap,
      incomeGapPercentage: (gapPercentage * 100).toFixed(1),
      totalGuaranteedIncome,
      planningHorizon,
      recommendedAmount: Math.round(recommendedAmount),
      recommendedPercentage: recommendedPercentage.toFixed(1),
      estimatedAnnualIncome: Math.round(estimatedAnnualIncome),
      payoutRate: (payoutRate * 100).toFixed(1),
      maxAnnuityAmount: Math.round(maxAnnuityAmount),
      investableAssets: Math.round(investableAssets),
    };
  };

  const generatePDF = (calculatedResults) => {
    const doc = new jsPDF();
    const pageWidth = doc.internal.pageSize.getWidth();
    const pageHeight = doc.internal.pageSize.getHeight();
    let yPos = 20;

    // Header
    doc.setFontSize(20);
    doc.text('Gambit Capital Management', 20, yPos);
    doc.setFontSize(14);
    doc.text('Annuity Suitability Analysis', 20, yPos + 10);

    yPos += 30;

    // Suitability Score
    doc.setFontSize(28);
    doc.setTextColor(0, 153, 102);
    doc.text(String(calculatedResults.totalScore), 20, yPos);
    doc.setTextColor(0, 0, 0);
    doc.setFontSize(12);
    doc.text('Suitability Score (0–100)', 20, yPos + 10);

    yPos += 25;

    // Score Breakdown
    doc.setFontSize(11);
    doc.text('Score Components:', 20, yPos);
    yPos += 8;
    doc.setFontSize(10);
    const scoreItems = [
      `Longevity (${calculatedResults.planningHorizon}-year horizon): ${calculatedResults.longevityScore}/25`,
      `Income Gap (${calculatedResults.incomeGapPercentage}% shortfall): ${calculatedResults.incomeGapScore}/25`,
      `Flexibility Need: ${calculatedResults.flexibilityScore}/25`,
      `Behavioral Fit: ${calculatedResults.behavioralScore}/25`
    ];
    scoreItems.forEach(item => {
      doc.text(item, 25, yPos);
      yPos += 7;
    });

    yPos += 8;

    // Recommendation
    doc.setFontSize(11);
    doc.text('Right-Sized Recommendation:', 20, yPos);
    yPos += 8;
    doc.setFontSize(10);
    const recItems = [
      `Recommended annuity amount: $${calculatedResults.recommendedAmount.toLocaleString()}`,
      `As percentage of investable assets: ${calculatedResults.recommendedPercentage}%`,
      `Estimated annual income: $${calculatedResults.estimatedAnnualIncome.toLocaleString()}/year`,
      `Based on payout rate: ${calculatedResults.payoutRate}% at age ${formData.age}`
    ];
    recItems.forEach(item => {
      doc.text(item, 25, yPos);
      yPos += 7;
    });

    yPos += 8;

    // Key considerations
    if (yPos > pageHeight - 40) {
      doc.addPage();
      yPos = 20;
    }
    
    doc.setFontSize(11);
    doc.text('Key Considerations:', 20, yPos);
    yPos += 8;
    doc.setFontSize(10);

    const considerations = [];
    if (calculatedResults.flexibilityScore < 10) {
      considerations.push('• Heir priority or healthcare concern limits suitability');
    }
    if (calculatedResults.incomeGap > 0) {
      considerations.push(`• Annual income gap of $${Math.round(calculatedResults.incomeGap).toLocaleString()} exists`);
    }
    if (calculatedResults.recommendedPercentage > 30) {
      considerations.push('• Annuity allocation is significant; ensure adequate liquid reserves remain');
    }
    if (calculatedResults.behavioralScore < 13) {
      considerations.push('• Market volatility response suggests caution with annuity lock-in');
    }

    considerations.forEach(item => {
      doc.text(item, 25, yPos);
      yPos += 7;
    });

    yPos += 8;

    // Important disclaimers
    if (yPos > pageHeight - 50) {
      doc.addPage();
      yPos = 20;
    }

    doc.setFontSize(11);
    doc.setTextColor(100, 0, 0);
    doc.text('Important Disclosures:', 20, yPos);
    doc.setTextColor(0, 0, 0);
    yPos += 8;
    doc.setFontSize(9);

    const disclosures = [
      '• Annuities are permanent purchases. You cannot recover capital or access the underlying assets.',
      '• Inflation erodes purchasing power over decades. A payout today loses ~50% value in 25 years.',
      '• Annuity payments do not pass to heirs. Proceeds revert to the insurer upon death.',
      '• This analysis does not evaluate insurer credit quality or default risk.',
      '• Consult a financial advisor before making any annuity purchase decision.'
    ];

    disclosures.forEach(item => {
      const lines = doc.splitTextToSize(item, pageWidth - 40);
      lines.forEach(line => {
        if (yPos > pageHeight - 20) {
          doc.addPage();
          yPos = 20;
        }
        doc.text(line, 25, yPos);
        yPos += 6;
      });
    });

    return doc;
  };

    const calculated = calculateScore();
    setResults(calculated);
    setStep(3);
  };

        <header className="tool-header">
          <h1>Gambit Capital Management</h1>
        </header>
        
        <div className="step-indicator">Step 1 of 3</div>
        
        <div className="navigation">
          <button onClick={() => setStep(0)} className="back-link">← Back</button>

        <div className="step-content">
          <h2>Income & Assets</h2>
          
          <div className="form-group">
            <label>Your current age</label>

              onChange={handleChange}
              min="40"
              max="100"
            />
          </div>

          <div className="form-section">
            <h3>Annual Income You Currently Receive or Expect to Receive</h3>
            
            <div className="form-group">
              <label>Employment income (W-2 wages)</label>
              <div className="input-prefix">$</div>
              <input
                type="number"
                name="employmentIncome"
                value={formData.employmentIncome}
                onChange={handleChange}
                min="0"
              />
            </div>

            <div className="form-group">
              <label>Self-employment income</label>
              <div className="input-prefix">$</div>
              <input
                type="number"
                name="selfEmploymentIncome"
                value={formData.selfEmploymentIncome}
                onChange={handleChange}
                min="0"
              />
            </div>

            <div className="form-group">
              <label>Social Security annual benefit</label>
              <div className="input-prefix">$</div>
              <input
                type="number"
                name="socialSecurityAnnual"
                value={formData.socialSecurityAnnual}
                onChange={handleChange}
                min="0"
              />
            </div>

            <div className="form-group">
              <label>Pension or other guaranteed income (annual)</label>
              <div className="input-prefix">$</div>
              <input
                type="number"
                name="pensionAnnual"
                value={formData.pensionAnnual}
                onChange={handleChange}

              />
            </div>

              <div className="input-prefix">$</div>

                onChange={handleChange}

              />

        </div>

        <style>{`

          }

          .next-button:hover {
            background: #007755;
          }
        `}</style>
      </div>
    );
  }

  // STEP 2: Personal Factors
  if (step === 2) {
    return (
      <div className="tool-container">
        <header className="tool-header">
          <h1>Gambit Capital Management</h1>
        </header>
        
        <div className="step-indicator">Step 2 of 3</div>
        
        <div className="navigation">
          <button onClick={() => setStep(1)} className="back-link">← Back</button>
          <button onClick={() => { setStep(1); setResults(null); }} className="start-over-link">Start over</button>
        </div>

        <div className="step-content">
          <h2>Personal Factors</h2>
          
          <div className="form-group">
            <label>How old do you expect to live to?</label>
            <input
              type="number"
              name="lifeExpectancy"
              value={formData.lifeExpectancy}
              onChange={handleChange}
              min="70"
              max="110"
            />
            <small>Use a realistic estimate; avoid extremes. 85–92 is typical.</small>
          </div>

          <div className="form-group">
            <label>How important is it to leave money to heirs?</label>
            <select

              value={formData.heirsImportance}
              onChange={handleChange}
            >
              <option value="low">Not important</option>
              <option value="moderate">Somewhat important</option>
              <option value="high">Very important</option>

          </div>

          <div className="form-group">
            <label>Are you concerned about accessing assets for unexpected healthcare costs?</label>
            <select

          <div className="form-group">
            <label>When markets drop 20%, your instinct is to:</label>
            <select
