/* -----------------------------------------------------------------------------
   3. PLAIN-ENGLISH BREAKDOWN LINES
   Replaces scoring notation like "-10 for heirs priority, -10 for healthcare"
   -------------------------------------------------------------------------- */

export function flexibilityPlainText({ heirsPriority, healthcareConcern }) {
  if (heirsPriority && healthcareConcern) {
    return "You told us leaving money to family and covering healthcare costs both matter, so most of your savings needs to stay accessible.";
  }
  if (heirsPriority) {
    return "You want to leave money to family. An annuity pays you, not your heirs, so that limits how much belongs in one.";
  }
  if (healthcareConcern) {
    return "You expect meaningful healthcare costs. Money in an annuity can't be pulled out for a large medical bill, so more needs to stay liquid.";
  }
  return "You don't have competing claims on this money, so more of it can go toward guaranteed income.";
}

export function longevityPlainText({ expectedAge, horizonYears }) {
  if (horizonYears >= 35) {
    return `Planning to age ${expectedAge} means this money may need to last ${horizonYears} more years. That's a long time for savings to hold up on their own.`;
  }
  if (horizonYears >= 25) {
    return `Planning to age ${expectedAge} gives you a ${horizonYears}-year horizon. Long enough that outliving your savings is a real risk worth addressing.`;
  }
  return `Planning to age ${expectedAge} gives you a ${horizonYears}-year horizon. Shorter horizons favor keeping money flexible rather than locking it up.`;
}

export function incomeGapPlainText({ guaranteed, spendingGoal }) {
  const gap = Math.max(0, spendingGoal - guaranteed);
  const pct = spendingGoal > 0 ? Math.round((gap / spendingGoal) * 100) : 0;
  if (gap === 0) {
    return "Social Security and pension income already cover your spending goal. There's no gap for an annuity to fill.";
  }
  return `Social Security and pension income cover ${
    100 - pct
  }% of what you plan to spend. The other ${pct}%, about $${gap.toLocaleString()} a year, has to come out of savings.`;
}
